## the generated route module (notes)
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_starter/features/notes/presentation/screens/notes_screen.dart';
import 'package:go_router/go_router.dart';

/// Path of the Notes screen.
///
/// Move this and the route name below into `AppRoutes`
/// (`lib/core/routing/app_routes.dart`) when you register the route, so
/// navigation uses the same constants as every other screen.
const String notesRoutePath = '/notes';

/// Name of the Notes route.
const String notesRouteName = 'notes';

/// Builds the route of the screen, reachable once signed in.
///
/// Nothing registers it yet. Add a call to this function to `buildAppRoutes`
/// in `lib/core/routing/routes_registry.dart`: a route that is not registered
/// there is a 404.
GoRoute buildNotesRoute(Ref ref) {
  return GoRoute(
    path: notesRoutePath,
    name: notesRouteName,
    builder: (context, state) =>
        NotesScreen(id: state.uri.queryParameters['id'] ?? '1'),
  );
}
