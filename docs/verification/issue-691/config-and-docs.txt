# Dependabot version updates: weekly PRs for pub packages and GitHub Actions.
#
# Minor and patch bumps are grouped into one PR per ecosystem to keep the PR
# count down; majors still arrive one per package, since those are the ones
# that need reading. Every PR runs the Quality gate like any other.
#
# `update-types` is judged on the package Dependabot is updating, not on the
# packages pub has to move with it. A minor flutter_riverpod bump can need
# riverpod_generator, riverpod_annotation and freezed across a major, and
# would carry them into a group named minor-and-patch. Those packages
# version together, so they get their own group, listed first (a package
# joins the first group it matches) and taking any update type: their PR
# says what it is, and a major there means regenerating the committed
# *.g.dart / *.freezed.dart files.
#
# A 0.x minor (0.13 -> 0.14) counts as minor here, though pub treats it as
# breaking; read those in the grouped PR.
#
# This file ships with the template, so a repository created from it gets the
# same PRs. To turn them off, delete this file (or one `updates:` entry);
# security updates are a separate setting under Settings -> Code security.
version: 2

updates:
  - package-ecosystem: pub
    directory: /
    schedule:
      interval: weekly
    open-pull-requests-limit: 5
    commit-message:
      prefix: chore(deps)
    groups:
      pub-codegen:
        patterns:
          - build_runner
          - flutter_riverpod
          - riverpod*
          - freezed*
          - json_annotation
          - json_serializable
      pub-minor-and-patch:
        update-types: [minor, patch]
    ignore:
      # `patrol` must move together with the `patrol_cli` version pinned in
      # .github/workflows/e2e-android.yml (see the compatibility table linked
      # there); a lone bump makes the CLI abort before any test runs. Bump
      # both by hand.
      - dependency-name: patrol

  - package-ecosystem: github-actions
